cmake_minimum_required(VERSION 3.20)
project(rtc_log_sdk VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(LOGCPP_SOURCE_DIR "" CACHE PATH "User-provided log-cpp checkout; never downloaded")
if(NOT EXISTS "${LOGCPP_SOURCE_DIR}/inc/w_log_cpp.h")
  message(FATAL_ERROR "LOGCPP_SOURCE_DIR must contain the user-provided source")
endif()
if(NOT RTC_LOG_SOURCE_REVISION)
  message(FATAL_ERROR "The packaging script must supply the verified source revision")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(RTC_LOG_PLATFORM linux-aarch64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  set(RTC_LOG_PLATFORM linux-x86_64)
else()
  message(FATAL_ERROR "This SDK recipe currently supports Linux ARM64 and x86_64 only")
endif()
set(LOGCPP_BUILD_SHARED OFF CACHE BOOL "Private static implementation" FORCE)
set(LOGCPP_SUPPORT_SQLITE OFF CACHE BOOL "No SQLite" FORCE)
set(LOGCPP_BUILD_TESTS OFF CACHE BOOL "SDK only" FORCE)
set(LOGCPP_BUILD_QT_TEST OFF CACHE BOOL "No Qt dependency" FORCE)
add_subdirectory("${LOGCPP_SOURCE_DIR}" log-cpp EXCLUDE_FROM_ALL)
add_library(rtc_log_adapter SHARED src/adapter.cpp)
target_include_directories(rtc_log_adapter PUBLIC include)
target_link_libraries(rtc_log_adapter PRIVATE log-cpp::log-cpp)
target_compile_definitions(rtc_log_adapter PRIVATE RTC_LOG_BUILD)
target_compile_options(rtc_log_adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(rtc_log_adapter PROPERTIES
  VERSION 1.0.0 SOVERSION 1 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_options(rtc_log_adapter PRIVATE
  "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
  "LINKER:-z,defs" "LINKER:-z,relro" "LINKER:-z,now")
set_property(TARGET rtc_log_adapter APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
install(TARGETS rtc_log_adapter LIBRARY DESTINATION "lib/${RTC_LOG_PLATFORM}" COMPONENT rtc_log_sdk)
install(FILES include/rtc_log_adapter.h DESTINATION include COMPONENT rtc_log_sdk)
foreach(dependency fmt spdlog)
  install(FILES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${dependency}/copyright"
    DESTINATION licenses RENAME "${dependency}.txt" COMPONENT rtc_log_sdk)
endforeach()
configure_file(cmake/manifest.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/manifest.cmake" @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/manifest.cmake" COMPONENT rtc_log_sdk)
