cmake_minimum_required(VERSION 3.24)
project(rtc_talk_client VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(TALK_WITH_WEBRTC "构建原生媒体引擎" OFF)
option(TALK_WITH_QT "构建 Qt5 界面" OFF)
include(CTest)
if(CMAKE_CROSSCOMPILING AND VCPKG_INSTALLED_DIR)
  list(APPEND CMAKE_FIND_ROOT_PATH "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
endif()
find_package(Threads REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(asio CONFIG REQUIRED)
find_package(websocketpp CONFIG REQUIRED)
add_library(talk_protocol INTERFACE)
target_include_directories(talk_protocol INTERFACE core)
target_link_libraries(talk_protocol INTERFACE nlohmann_json::nlohmann_json)
add_library(talk_core SHARED core/core.cpp)
target_include_directories(talk_core PUBLIC include PRIVATE core)
target_link_libraries(talk_core PRIVATE talk_protocol asio::asio websocketpp::websocketpp Threads::Threads)
target_compile_definitions(talk_core PRIVATE TALK_BUILDING_LIBRARY ASIO_STANDALONE)
target_compile_options(talk_core PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
set_target_properties(talk_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_options(talk_core PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/core/exports.map" "LINKER:--gc-sections" "LINKER:-z,defs")
set_property(TARGET talk_core APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/core/exports.map")
if(TALK_WITH_WEBRTC)
  include(cmake/webrtc.cmake)
  target_sources(talk_core PRIVATE core/media/webrtc_media.cpp)
  target_compile_definitions(talk_core PRIVATE TALK_WITH_WEBRTC)
  set_source_files_properties(core/media/webrtc_media.cpp PROPERTIES COMPILE_OPTIONS "-fno-rtti")
  target_link_libraries(talk_core PRIVATE talk_webrtc)
else()
  target_sources(talk_core PRIVATE core/media/null_media.cpp)
endif()
add_executable(talk-cli apps/talk-cli/main.cpp)
target_link_libraries(talk-cli PRIVATE talk_core nlohmann_json::nlohmann_json Threads::Threads)
set_target_properties(talk-cli PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
if(TALK_WITH_QT)
  # 预先导入宿主工具，避免 SDK 中旧的绝对路径造成误解析。
  if(CMAKE_CROSSCOMPILING)
    foreach(tool qmake moc rcc uic)
      add_executable(Qt5::${tool} IMPORTED)
      set_target_properties(Qt5::${tool} PROPERTIES IMPORTED_LOCATION "${RK3576_HOST}/bin/${tool}")
    endforeach()
  endif()
  find_package(Qt5 5.15 COMPONENTS Core Gui Widgets Network REQUIRED)
  add_executable(talk-qt apps/talk-qt/main.cpp ui/qt/window.cpp ui/qt/window.h)
  set_target_properties(talk-qt PROPERTIES AUTOMOC ON INSTALL_RPATH "$ORIGIN/../lib")
  target_include_directories(talk-qt PRIVATE ui/qt)
  target_link_libraries(talk-qt PRIVATE talk_core Qt5::Widgets Qt5::Network)
  install(TARGETS talk-qt RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
  add_executable(api-test tests/api_test.c)
  target_link_libraries(api-test PRIVATE talk_core)
  add_test(NAME api COMMAND api-test)
  set_tests_properties(api PROPERTIES TIMEOUT 15)
  add_executable(protocol-test tests/protocol_test.cpp)
  target_link_libraries(protocol-test PRIVATE talk_protocol)
  add_test(NAME protocol COMMAND protocol-test)
  add_executable(signaling-test tests/signaling_test.cpp)
  # 锁定版 websocketpp 的服务端测试夹具仍使用 C++17；产品客户端保持 C++20。
  set_target_properties(signaling-test PROPERTIES CXX_STANDARD 17)
  target_compile_definitions(signaling-test PRIVATE ASIO_STANDALONE)
  target_link_libraries(signaling-test PRIVATE talk_core talk_protocol asio::asio websocketpp::websocketpp Threads::Threads)
  add_test(NAME signaling COMMAND signaling-test)
  set_tests_properties(signaling PROPERTIES TIMEOUT 45)
endif()
install(TARGETS talk_core talk-cli LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/talk/talk.h DESTINATION include/talk)
