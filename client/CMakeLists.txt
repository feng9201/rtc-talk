cmake_minimum_required(VERSION 3.20)
project(rtc_rk_client VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Threads REQUIRED)
option(RTC_WITH_SIGNALING "Build real WS/WSS protocol, logging and diagnostic CLI" OFF)
option(RTC_WITH_WEBRTC "Build RK libwebrtc SFU Opus backend" OFF)
option(RTC_ENABLE_TEST_AUDIO "Include synthetic audio device (development builds ONLY)" OFF)
if(RTC_WITH_WEBRTC AND NOT RTC_WITH_SIGNALING)
  message(FATAL_ERROR "RTC_WITH_WEBRTC requires RTC_WITH_SIGNALING")
endif()

option(RTC_ENABLE_SANITIZERS "Enable address/undefined sanitizers on host" OFF)
option(RTC_ENABLE_TSAN "Enable thread sanitizer on host (separate build)" OFF)
if(RTC_ENABLE_SANITIZERS AND RTC_ENABLE_TSAN)
  message(FATAL_ERROR "ASan and TSan must use separate builds")
endif()
if((RTC_ENABLE_SANITIZERS OR RTC_ENABLE_TSAN) AND
   (CMAKE_CROSSCOMPILING OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang"))
  message(FATAL_ERROR "This sanitizer preset is supported only on a GNU/Clang host")
endif()

add_library(rtc_instrumentation INTERFACE)
if(RTC_ENABLE_SANITIZERS)
  target_compile_options(rtc_instrumentation INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rtc_instrumentation INTERFACE -fsanitize=address,undefined)
elseif(RTC_ENABLE_TSAN)
  target_compile_options(rtc_instrumentation INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(rtc_instrumentation INTERFACE -fsanitize=thread)
endif()

add_library(rtc_foundation STATIC
  libs/domain/session.cpp
  libs/domain/operations.cpp
  libs/video/frame.cpp
  libs/video/budget.cpp
)
target_include_directories(rtc_foundation PUBLIC include)
target_link_libraries(rtc_foundation PUBLIC Threads::Threads rtc_instrumentation)
target_compile_options(rtc_foundation PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(BUILD_TESTING)
  foreach(suite session operations video concurrency)
    add_executable(rtc_${suite}_test tests/unit/${suite}_test.cpp)
    target_include_directories(rtc_${suite}_test PRIVATE tests/unit)
    target_compile_options(rtc_${suite}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(rtc_${suite}_test PRIVATE rtc_foundation)
    add_test(NAME ${suite} COMMAND rtc_${suite}_test)
    set_tests_properties(${suite} PROPERTIES TIMEOUT 20)
  endforeach()
endif()

if(RTC_WITH_SIGNALING)
  find_package(nlohmann_json CONFIG REQUIRED)
  find_package(asio CONFIG REQUIRED)
  find_package(websocketpp CONFIG REQUIRED)
  include(cmake/log-sdk.cmake)
  # TLS comes from the selected platform SDK, not another business workspace.
  find_package(OpenSSL 1.1.1 REQUIRED)
  message(STATUS "Platform TLS: ${OPENSSL_VERSION}; security qualification remains a release gate")
  add_library(rtc_protocol STATIC libs/protocol/message.cpp)
  target_include_directories(rtc_protocol PUBLIC include)
  target_link_libraries(rtc_protocol PUBLIC nlohmann_json::nlohmann_json)
  add_library(rtc_logging STATIC libs/logging/logger.cpp)
  target_link_libraries(rtc_logging PUBLIC rtc_protocol PRIVATE rtc_log_sdk Threads::Threads)
  add_library(rtc_signaling STATIC libs/signaling/rpc.cpp libs/signaling/ws_transport.cpp)
  target_link_libraries(rtc_signaling PUBLIC rtc_protocol asio::asio
    PRIVATE rtc_logging websocketpp::websocketpp OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
  target_compile_definitions(rtc_signaling PUBLIC ASIO_STANDALONE)
  add_library(rtc_media_worker STATIC libs/media/worker.cpp)
  target_link_libraries(rtc_media_worker PUBLIC rtc_foundation rtc_protocol asio::asio)
  target_compile_definitions(rtc_media_worker PUBLIC ASIO_STANDALONE)
  foreach(target rtc_protocol rtc_logging rtc_signaling rtc_media_worker)
    target_link_libraries(${target} PUBLIC rtc_instrumentation)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
  add_executable(rtc-cli apps/rtc-cli/main.cpp)
  set_target_properties(rtc-cli PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  target_link_libraries(rtc-cli PRIVATE rtc_signaling rtc_logging rtc_foundation rtc_media_worker)
  if(RTC_WITH_WEBRTC)
    include(cmake/webrtc.cmake)
    # Keep WebRTC/BoringSSL out of the executable's OpenSSL link namespace.
    # A static archive in rtc-cli allowed OpenSSL to satisfy BoringSSL references.
    add_library(rtc_audio SHARED libs/media/webrtc_audio.cpp)
    set_target_properties(rtc_audio PROPERTIES CXX_STANDARD 20
      CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
      VERSION 0.1.0 SOVERSION 0 INSTALL_RPATH "$ORIGIN")
    target_compile_options(rtc_audio PRIVATE -Wall -Wextra -fno-rtti)
    target_link_libraries(rtc_audio PRIVATE rtc_webrtc rtc_logging PUBLIC rtc_media_worker)
    # Cross ld must also resolve the DSO's private log SDK dependency; do not
    # expose private headers or bake this host path into installed RPATH.
    target_link_options(rtc_audio INTERFACE
      "LINKER:-rpath-link,${RTC_LOG_SDK_ROOT}/lib/${_rtc_log_platform}")
    target_link_options(rtc_audio PRIVATE
      "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/libs/media/exports.map"
      "LINKER:--exclude-libs,ALL" "LINKER:-Bsymbolic" "LINKER:-z,defs")
    set_property(TARGET rtc_audio APPEND PROPERTY LINK_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/libs/media/exports.map")
    add_custom_command(TARGET rtc_audio POST_BUILD
      COMMAND ${CMAKE_COMMAND} "-DMEDIA_LIBRARY=$<TARGET_FILE:rtc_audio>" "-DREADELF=${CMAKE_READELF}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/check-media-linkage.cmake"
      VERBATIM)
    install(TARGETS rtc_audio LIBRARY DESTINATION lib)
    target_compile_definitions(rtc-cli PRIVATE RTC_WITH_WEBRTC)
    target_link_libraries(rtc-cli PRIVATE rtc_audio)
    if(RTC_ENABLE_TEST_AUDIO)
      target_compile_definitions(rtc_audio PRIVATE RTC_ENABLE_TEST_AUDIO)
      target_compile_definitions(rtc-cli PRIVATE RTC_ENABLE_TEST_AUDIO)
      if(BUILD_TESTING)
        add_executable(rtc_audio_backend_test tests/integration/audio_backend_test.cpp)
        target_include_directories(rtc_audio_backend_test PRIVATE tests/unit)
        target_link_libraries(rtc_audio_backend_test PRIVATE rtc_audio)
        add_test(NAME audio_backend COMMAND rtc_audio_backend_test)
        set_tests_properties(audio_backend PROPERTIES TIMEOUT 25)
        # Regression: exercise real DTLS certificate creation while a live
        # system-OpenSSL TLS context exists in the same process.
        add_executable(rtc_audio_tls_test tests/integration/audio_backend_test.cpp)
        target_include_directories(rtc_audio_tls_test PRIVATE tests/unit)
        target_compile_definitions(rtc_audio_tls_test PRIVATE RTC_TEST_WITH_TLS ASIO_STANDALONE)
        target_link_libraries(rtc_audio_tls_test PRIVATE rtc_audio asio::asio OpenSSL::SSL OpenSSL::Crypto)
        add_test(NAME audio_tls COMMAND rtc_audio_tls_test)
        set_tests_properties(audio_tls PROPERTIES TIMEOUT 25)
      endif()
    endif()
  endif()
  target_compile_options(rtc-cli PRIVATE -Wall -Wextra -Wpedantic -Werror)
  install(TARGETS rtc-cli RUNTIME DESTINATION bin)
  install(FILES "${_rtc_log_library}" DESTINATION lib RENAME librtc_log_adapter.so.1)
  if(BUILD_TESTING)
    add_executable(rtc_media_worker_test tests/unit/media_worker_test.cpp)
    target_include_directories(rtc_media_worker_test PRIVATE tests/unit)
    target_link_libraries(rtc_media_worker_test PRIVATE rtc_media_worker)
    target_compile_options(rtc_media_worker_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME media_worker COMMAND rtc_media_worker_test)
    set_tests_properties(media_worker PROPERTIES TIMEOUT 20)
    add_executable(rtc_log_abi_test tests/unit/log_abi_test.c)
    target_link_libraries(rtc_log_abi_test PRIVATE rtc_log_sdk rtc_instrumentation)
    target_compile_options(rtc_log_abi_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME log_abi COMMAND rtc_log_abi_test)
    set_tests_properties(log_abi PROPERTIES TIMEOUT 20)
    foreach(suite protocol rpc logging)
      add_executable(rtc_${suite}_test tests/unit/${suite}_test.cpp)
      target_include_directories(rtc_${suite}_test PRIVATE tests/unit)
      target_link_libraries(rtc_${suite}_test PRIVATE rtc_signaling rtc_logging)
      target_compile_options(rtc_${suite}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
      add_test(NAME ${suite} COMMAND rtc_${suite}_test)
      set_tests_properties(${suite} PROPERTIES TIMEOUT 20)
    endforeach()
    target_link_libraries(rtc_logging_test PRIVATE rtc_log_sdk)
    add_executable(rtc_transport_test tests/integration/transport_test.cpp)
    target_include_directories(rtc_transport_test PRIVATE tests/unit tests/integration)
    target_link_libraries(rtc_transport_test PRIVATE rtc_signaling websocketpp::websocketpp OpenSSL::SSL OpenSSL::Crypto)
    if(RTC_WITH_WEBRTC AND RTC_ENABLE_TEST_AUDIO)
      target_compile_definitions(rtc_transport_test PRIVATE RTC_TEST_WITH_WEBRTC)
      target_link_libraries(rtc_transport_test PRIVATE rtc_audio)
    endif()
    add_test(NAME transport COMMAND rtc_transport_test)
    set_tests_properties(transport PROPERTIES TIMEOUT 20)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_test.py $<TARGET_FILE:rtc-cli>)
    set_tests_properties(cli PROPERTIES TIMEOUT 15)
  endif()
endif()
